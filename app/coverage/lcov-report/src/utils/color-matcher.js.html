function hexToRgb(hex) {
  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
  return result ? [
    parseInt(result[1], 16),
    parseInt(result[2], 16),
    parseInt(result[3], 16)
  ] : [0,0,0];
}
 
// Convert RGB to XYZ (needed for LAB)
function rgbToXyz(r, g, b) {
  let [rL, gL, bL] = [r, g, b].map(v => {
    v /= 255;
    return v > 0.04045 ? Math.pow((v + 0.055) / 1.055, 2.4) : v / 12.92;
  });
  
  rL *= 100;
  gL *= 100;
  bL *= 100;
  
  const x = rL * 0.4124 + gL * 0.3576 + bL * 0.1805;
  const y = rL * 0.2126 + gL * 0.7152 + bL * 0.0722;
  const z = rL * 0.0193 + gL * 0.1192 + bL * 0.9505;
  
  return [x, y, z];
}
 
// Convert XYZ to CIE LAB
function xyzToLab(x, y, z) {
  let [xL, yL, zL] = [x / 95.047, y / 100.000, z / 108.883].map(v => 
    v > 0.008856 ? Math.pow(v, 1/3) : (7.787 * v) + (16 / 116)
  );
  
  return [
    (116 * yL) - 16,
    500 * (xL - yL),
    200 * (yL - zL)
  ];
}
 
function rgbToLab(r, g, b) {
  return xyzToLab(...rgbToXyz(r, g, b));
}
 
// Basic CIE 1976 Euclidean Delta E distance
function deltaE(lab1, lab2) {
  return Math.sqrt(
    Math.pow(lab1[0] - lab2[0], 2) + 
    Math.pow(lab1[1] - lab2[1], 2) + 
    Math.pow(lab1[2] - lab2[2], 2)
  );
}
 
// Cache LAB values directly onto the dictionary for blazing fast subsequent lookups
COLOR_DICTIONARY.forEach(c => {
  const [r, g, b] = hexToRgb(c.hex);
  c.lab = rgbToLab(r, g, b);
});
 
/**
 * Compare target RGB to the CSS color dictionary using LAB perceptual distance
 */
export function getClosestColorName(r, g, b) {
  const targetLab = rgbToLab(r, g, b);
  
  let closestName = 'black';
  let minDistance = Infinity;
 
  for (const color of COLOR_DICTIONARY) {
    const distance = deltaE(targetLab, color.lab);
    if (distance < minDistance) {
      minDistance = distance;
      closestName = color.name;
    }
  }
 
  return closestName;
}
